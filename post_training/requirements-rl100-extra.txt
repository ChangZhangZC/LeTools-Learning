# ARBiM RL-100 incremental dependencies.
# Run from the LeTools-Learning repository root:
#   python -m pip install --no-build-isolation -r post_training/requirements-rl100-extra.txt
#
# Add new RL-100 integration dependencies here as we encounter/validate them.
# Do not duplicate LeTools' base requirements or copy RL-100's full environment.
#
# PyTorch3D: use RL-100's bundled minimal *real* extension, not a fake stub.
# Source version is pinned by the RL-100 git submodule (6b6afceb7eb12525398207b5d6e66908863dc385).
# Requires a working PyTorch installation and C++ build toolchain;
# CUDA compilation additionally needs a compatible CUDA toolkit/nvcc.
-e ./third_party/rl100/third_party/pytorch3d_simplified

# Legacy 'import gym' used by RL-100 unidpg.utils (distinct from LeTools gymnasium).
# Gym 0.26.2 is selected as a Python 3.12-oriented import bridge, NOT as an
# endorsement of its Gym 0.21/old-step-API behavior for online environments.
# Online RL will use a separately validated environment adapter.
gym==0.26.2

# Zarr v2 API is used by the pinned RL-100 LinearNormalizer and offline datasets.
# Use a Python 3.12-compatible Zarr 2.x release, not the incompatible v3 API.
# Pin numcodecs below 0.16: newer versions removed cbuffer_sizes used by Zarr v2.
zarr==2.18.7
numcodecs==0.15.1

# Imported by rl_100.policy.rl100_2d for debug plots / image visualization.
# Python 3.12 compatible range; pin the tested server version in a later lockfile.
matplotlib>=3.8,<4

# Original kdc_ARBiM SequenceSampler uses @numba.jit for episode-safe indices.
# Numba 0.61.2+ supports Python 3.12 and NumPy 2.2; <0.63 bounds this
# migration's initial compatibility range (exact server version to be locked).
numba>=0.61.2,<0.63
